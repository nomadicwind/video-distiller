```ts
export const GUTTER_W = 112      // 轨道头沟槽宽
export const RULER_H = 28        // 标尺高（替代 TOP_H=16；TOP_H 保留导出=RULER_H 以兼容）
export const LANE_H = 72         // 由 64 调至 72
export const TOOLBAR_H = 28

export function niceTickInterval(spanMs: number, widthPx: number): { majorMs: number; minorMs: number }
// 目标主刻度间距 ≥80px：从 [1,2,5]×10^n (n≥0) 序列取最小满足 majorMs/spanMs*widthPx >= 80 者；minorMs = majorMs/5

export function snapMs(rawMs: number, fps: number, magnets: number[], v: Viewport, tolPx?: number): number
// 先帧取整 Math.round(rawMs/frame)*frame（frame=1000/fps）；再若与任一 magnet 的像素距离 <= tolPx(默认6) 则取该 magnet；magnets 为打表 t_ms + 相邻标记 t_ms 集合
```

注意：`msToPx/pxToMs` 语义不变，但从任务 5 起绘制/命中区在 x 方向整体右移 GUTTER_W —— 约定：Viewport.widthPx 仍指**轨道内容区宽**（不含沟槽），调用方（Timeline.tsx）负责在事件坐标上先减 GUTTER_W。既有测试因此不需改。
